cmake_minimum_required(VERSION 3.10)
project(task_core_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS agx_task_workflow)
add_executable(task_core_consumer main.cpp)
target_include_directories(task_core_consumer PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(task_core_consumer ${catkin_LIBRARIES})
target_compile_options(task_core_consumer PRIVATE -UNDEBUG)
