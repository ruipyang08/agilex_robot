cmake_minimum_required(VERSION 3.10)
project(agx_task_workflow)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_path(yaml-cpp_INCLUDE_DIRS yaml-cpp/yaml.h HINTS ${YAML_CPP_INCLUDE_DIR})
set(yaml-cpp_LIBRARIES ${YAML_CPP_LIBRARIES})
catkin_package(INCLUDE_DIRS include LIBRARIES task_workflow ${CMAKE_THREAD_LIBS_INIT} DEPENDS yaml-cpp)
add_library(task_workflow src/config.cpp src/runner.cpp src/task_registry.cpp src/task_runtime_ports.cpp)
target_include_directories(task_workflow PUBLIC include ${yaml-cpp_INCLUDE_DIRS})
target_link_libraries(task_workflow PUBLIC yaml-cpp Threads::Threads)
add_executable(record_value_example examples/record_value/main.cpp)
target_link_libraries(record_value_example task_workflow)
if(CATKIN_ENABLE_TESTING)
  foreach(test config runner task_registry execution_context runtime_ports planning_state record_value)
    add_executable(task_${test}_test test/${test}_test.cpp)
    target_link_libraries(task_${test}_test task_workflow)
    target_compile_options(task_${test}_test PRIVATE -UNDEBUG)
    add_test(NAME task_${test}_test COMMAND task_${test}_test)
  endforeach()
  add_executable(task_core_consumer_test test/core_consumer/main.cpp)
  target_link_libraries(task_core_consumer_test task_workflow)
  target_compile_options(task_core_consumer_test PRIVATE -UNDEBUG)
  add_test(NAME task_core_consumer_test COMMAND task_core_consumer_test)
endif()
install(TARGETS task_workflow record_value_example
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY examples docs DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
