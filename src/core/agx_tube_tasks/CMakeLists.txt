cmake_minimum_required(VERSION 3.10)
project(agx_tube_tasks)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS agx_task_workflow agx_robot_task_runtime agx_xarm_pick)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_path(yaml-cpp_INCLUDE_DIRS yaml-cpp/yaml.h HINTS ${YAML_CPP_INCLUDE_DIR})
set(yaml-cpp_LIBRARIES ${YAML_CPP_LIBRARIES})
catkin_package(INCLUDE_DIRS include LIBRARIES tube_tasks CATKIN_DEPENDS agx_task_workflow agx_robot_task_runtime agx_xarm_pick DEPENDS yaml-cpp)
add_library(tube_tasks src/grasp_execution.cpp src/grasp_observation_json.cpp src/grasp_task.cpp src/grasp_test_tube_task.cpp src/task_config.cpp)
target_include_directories(tube_tasks PUBLIC include ${catkin_INCLUDE_DIRS})
add_dependencies(tube_tasks ${catkin_EXPORTED_TARGETS})
target_link_libraries(tube_tasks PUBLIC yaml-cpp Threads::Threads ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  foreach(test task_config grasp_effects grasp_observation_json grasp_journal_status)
    add_executable(${PROJECT_NAME}_${test}_test test/${test}_test.cpp)
    target_link_libraries(${PROJECT_NAME}_${test}_test tube_tasks)
    target_compile_options(${PROJECT_NAME}_${test}_test PRIVATE -UNDEBUG)
    add_test(NAME ${PROJECT_NAME}_${test}_test COMMAND ${PROJECT_NAME}_${test}_test)
  endforeach()
  set_tests_properties(agx_tube_tasks_grasp_effects_test
    agx_tube_tasks_grasp_observation_json_test agx_tube_tasks_grasp_journal_status_test
    PROPERTIES TIMEOUT 45)
endif()
install(TARGETS tube_tasks
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY docs DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
