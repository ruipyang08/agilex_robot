cmake_minimum_required(VERSION 3.10)
project(agx_navigation_tasks)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS agx_task_workflow agx_robot_task_runtime agx_xarm_pick roscpp std_msgs bt_task_msgs moveit_core moveit_msgs moveit_ros_planning_interface geometric_shapes)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_path(yaml-cpp_INCLUDE_DIRS yaml-cpp/yaml.h HINTS ${YAML_CPP_INCLUDE_DIR})
set(yaml-cpp_LIBRARIES ${YAML_CPP_LIBRARIES})
find_package(OpenSSL REQUIRED)
set(OpenSSL_INCLUDE_DIRS ${OPENSSL_INCLUDE_DIR})
set(OpenSSL_LIBRARIES ${OPENSSL_CRYPTO_LIBRARY})
catkin_package(INCLUDE_DIRS include LIBRARIES navigation_tasks CATKIN_DEPENDS agx_task_workflow agx_robot_task_runtime agx_xarm_pick roscpp std_msgs bt_task_msgs moveit_core moveit_msgs moveit_ros_planning_interface geometric_shapes DEPENDS yaml-cpp OpenSSL)
add_library(navigation_tasks src/navigate_task.cpp src/navigation_task.cpp src/task_config.cpp src/travel_envelope.cpp src/travel_envelope_ros.cpp src/travel_motion.cpp src/travel_policy.cpp src/travel_profile.cpp)
target_include_directories(navigation_tasks PUBLIC include ${catkin_INCLUDE_DIRS})
add_dependencies(navigation_tasks ${catkin_EXPORTED_TARGETS})
target_link_libraries(navigation_tasks PUBLIC yaml-cpp Threads::Threads ${catkin_LIBRARIES} OpenSSL::Crypto)
if(CATKIN_ENABLE_TESTING)
  foreach(test task_config navigation_task travel_profile travel_policy travel_retention bounded_motion travel_motion_sequence travel_envelope travel_spline)
    add_executable(${PROJECT_NAME}_${test}_test test/${test}_test.cpp)
    target_link_libraries(${PROJECT_NAME}_${test}_test navigation_tasks)
    target_compile_options(${PROJECT_NAME}_${test}_test PRIVATE -UNDEBUG)
    add_test(NAME ${PROJECT_NAME}_${test}_test COMMAND ${PROJECT_NAME}_${test}_test)
  endforeach()
endif()
install(TARGETS navigation_tasks
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY docs DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
